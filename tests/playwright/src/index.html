from playwright.sync_api import Page, expect def test_selectors(page: Page): page.goto("file:///C:/path/to/playwright-selectors.html") # Looks like a link, but is a button button = page.get_by_role("button", name="Open Details") expect(button).to_be_visible() button.click() expect(page.locator("#result")).to_have_text("Button clicked!") # Looks like a button, but is a link link = page.get_by_role("link", name="Go to Section") expect(link).to_be_visible() link.click() expect(page).to_have_url("#destination")
